-- ============================================================
-- 1. PLAYERS (No partition)
-- ============================================================
CREATE EXTERNAL TABLE IF NOT EXISTS silver_nfl_players (
    birthday DATE,
    college STRING,
    current_status STRING,
    current_team STRING,
    experience STRING,
    high_school_location STRING,
    player_id INT,
    last_name STRING,
    first_name STRING,
    birth_city STRING,
    birth_state STRING
)
STORED AS PARQUET
LOCATION '/tmp/DE011025/Joe/silver/nfl_players/';
